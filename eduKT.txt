    st.session_state.premium = False

    st.session_state.premium_pagado = False

    st.session_state.premium_expira = None

    st.session_state.premium_expira = (

if "temas_completados" not in st.session_state:
    st.session_state.temas_completados = set()


def guardar_perfil_actual():
    """Guarda el avance actual para recuperarlo al volver a iniciar sesión."""
    usuario = st.session_state.usuario_actual
    if usuario:
        perfil = st.session_state.perfiles[usuario]
        perfil.update(
            puntos=st.session_state.puntos,
            nivel=st.session_state.nivel,
            premium=st.session_state.premium,
            premium_pagado=st.session_state.premium_pagado,
            premium_expira=st.session_state.premium_expira,
            racha=st.session_state.racha,
            racha_temas=st.session_state.racha_temas,
            racha_respuestas=st.session_state.racha_respuestas,
            temas_completados=set(st.session_state.temas_completados),
        )


def cargar_perfil(usuario):
    """Carga el progreso guardado de un perfil seleccionado."""
    perfil = st.session_state.perfiles[usuario]
    st.session_state.usuario_actual = usuario
    for clave in (
        "puntos", "nivel", "premium", "premium_pagado", "premium_expira",
        "racha", "racha_temas", "racha_respuestas", "temas_completados",
    ):
        st.session_state[clave] = perfil[clave]

ETAPAS_NIVEL_ALTO = {
    "4° grado",
    "5° grado",
    "6° grado",
    "Ciclo preuniversitario",
}


def es_nivel_alto(nivel, etapa):
    """Distingue los grados de Primaria de los grados homónimos de Secundaria."""
    return (
        nivel in {"Secundaria", "Preuniversitario"}
        or nivel == "Primaria" and etapa in ETAPAS_NIVEL_ALTO
        or etapa == "Ciclo preuniversitario"
    )


def preparar_ejercicios_nivel_alto(ejercicios, area, subtema):
    """Convierte los ejercicios en casos extensos y exigentes, con respuesta concreta."""
    instrucciones = {
        "Matemática": (
            "Organiza los datos, resuelve las operaciones en el orden correcto y expresa el resultado con su unidad."
        ),
        "Comunicación": (
            "Relaciona la información explícita con una inferencia y fundamenta tu respuesta con un elemento del caso."
        ),
        "Ciencia, Tecnología y Ambiente (CIT)": (
            "Identifica variables, describe el dato que comprobarías y formula una conclusión basada en evidencia."
        ),
        "Ciencias Sociales": (
            "Relaciona actores, tiempo y territorio; distingue causas de consecuencias y sustenta la conclusión con una evidencia."
        ),
    }
    instruccion = instrucciones[area]
    return [
        f"Caso aplicado de {area} sobre {subtema}: analiza cuidadosamente la situación. "
        f"{ejercicio} {instruccion} Escribe al final únicamente la respuesta concreta solicitada."
        for ejercicio in ejercicios
    ]

def generar_banco_examen(area, subtema, etapa=None, nivel=None):
    """Devuelve 50 problemas variados, manteniendo tres niveles de dificultad."""
    if area == "Matemática" and nivel in {"Primaria", "Secundaria", "Preuniversitario"}:
        problemas = generar_matematica_avanzada(subtema, etapa, nivel)
    elif area == "Comunicación":
        problemas = generar_comunicacion_practica(subtema, etapa, nivel)
    elif area == "Ciencia, Tecnología y Ambiente (CIT)" and es_tema_fisica(subtema):
        problemas = generar_fisica_practica(subtema, etapa, nivel)
    elif area == "Ciencia, Tecnología y Ambiente (CIT)":
        problemas = generar_cyt_practica(subtema, etapa, nivel)
    elif subtema == "Números enteros":
        problemas = [
            f"Calcula: {indice} + (-{indice + 2})."
            for indice in range(1, 21)
        ]
        problemas += [
            f"Calcula: (-{indice + 3}) - (-{indice})."
            for indice in range(1, 21)
        ]
        problemas += [
            f"Una temperatura de -{indice} °C aumenta {indice + 5} °C. ¿Cuál es la temperatura final?"
            for indice in range(1, 21)
        ]
        problemas += [
            f"Calcula: (-{indice + 2}) × {indice} y determina el signo del resultado."
            for indice in range(1, 21)
        ]
        problemas += [
            f"Un ascensor está en el piso -{indice} y sube {indice + 4} pisos. ¿En qué piso termina?"
            for indice in range(1, 21)
        ]
    else:
        problemas = generar_banco_ejercicios(area, subtema, etapa, nivel)

    bloques = [problemas[indice:indice + 20] for indice in range(0, 100, 20)]
    seleccionados = []
    for bloque in bloques:
        seleccionados.extend(random.sample(bloque, 10))
    random.shuffle(seleccionados)
    if nivel != "Inicial":
        seleccionados = adaptar_problemas_al_tema(seleccionados, area, subtema, nivel)
    return agregar_opciones_examen(area, seleccionados)


def obtener_banco_examen(clave, area, subtema, etapa=None, nivel=None):
    """Conserva el mismo examen durante un intento y permite renovarlo explícitamente."""
    clave_examen = f"examen_{clave}"
    clave_version = f"{clave_examen}_version"
    if clave_examen not in st.session_state:
        st.session_state[clave_examen] = generar_banco_examen(area, subtema, etapa, nivel)
        st.session_state[clave_version] = st.session_state.get(clave_version, 0) + 1
    return st.session_state[clave_examen]


def obtener_version_examen(clave):
    """Devuelve la versión del intento para aislar las respuestas del formulario."""
    return st.session_state.get(f"examen_{clave}_version", 1)


def agregar_opciones_examen(area, problemas):
    """Convierte cada problema práctico en una pregunta con una respuesta correcta."""
    preguntas = []
    for indice, problema in enumerate(problemas, start=1):
        if area == "Matemática":
            opciones, correcta = opciones_matematica(problema, indice)
        elif area == "Comunicación":
            opciones, correcta = opciones_comunicacion(problema)
        elif area == "Ciencia, Tecnología y Ambiente (CIT)":
            opciones, correcta = opciones_ciencia(problema)
        else:
            opciones, correcta = opciones_sociales(problema)
        opciones = completar_cinco_opciones(opciones, correcta, area, problema)
        opciones = rotar_respuesta_correcta(opciones, correcta, indice)
        preguntas.append((problema, opciones, correcta))
    return preguntas


def completar_cinco_opciones(opciones, correcta, area, problema):
    """Añade dos distractores concretos para que cada pregunta tenga cinco opciones."""
    if area == "Matemática":
        valor = correcta.removeprefix("Resultado: ")
        try:
            numero = float(valor)
            extras = [f"Resultado: {numero + 2:g}", f"Resultado: {numero - 2:g}"]
        except ValueError:
            extras = ["Resultado: 0", "Resultado: 1"]
    elif area == "Comunicación":
        extras = [
            "Cambiar el propósito o los datos del mensaje sin justificarlo.",
            "Responder con una opinión que no se relaciona con el texto.",
        ]
    elif area == "Ciencia, Tecnología y Ambiente (CIT)":
        extras = [
            "Concluir sin registrar datos ni controlar las condiciones del caso.",
            "Elegir una explicación solo porque parece posible, sin comprobarla.",
        ]
    else:
        extras = [
            "Explicar el caso sin ubicarlo en su tiempo ni en su territorio.",
            "Usar una afirmación general sin relacionarla con la fuente o los actores.",

def opciones_cyt_practica(problema):
    """Calcula alternativas concretas para los casos prácticos de CIT."""
    densidad = re.search(r"muestra tiene (\d+) g y ocupa (\d+) cm³", problema)
    concentracion = re.search(r"disuelven (\d+) g de sustancia en (\d+) mL", problema)
    agua = re.search(r"familia consume (\d+) L.*?reduce (\d+) L", problema)
    residuos = re.search(r"recolectan (\d+) kg.*?reciclan (\d+) kg", problema)
    organismos = re.search(r"registran (\d+) organismos.*?y (\d+) sin ella", problema)
    cuidado = re.search(r"requiere (\d+) mL por.*?durante (\d+) días", problema)
    tratamientos = re.search(r"tratamiento A produce (\d+).*?tratamiento B (\d+)", problema)
    prototipo = re.search(r"necesitan (\d+) piezas y.*?cuesta S/ (\d+)", problema)
    if densidad:
        masa, volumen = map(float, densidad.groups())
        resultado, unidad = masa / volumen, "g/cm³"
    elif concentracion:
        soluto, volumen = map(float, concentracion.groups())
        resultado, unidad = soluto / volumen, "g/mL"
    elif agua:
        consumo, ahorro = map(float, agua.groups())
        resultado, unidad = consumo - ahorro, "L"
    elif residuos:
        recolectados, reciclados = map(float, residuos.groups())
        resultado, unidad = recolectados - reciclados, "kg"
    elif organismos:
        primer_grupo, segundo_grupo = map(float, organismos.groups())
        resultado, unidad = primer_grupo + segundo_grupo, "organismos"
    elif cuidado:
        dosis, dias = map(float, cuidado.groups())
        resultado, unidad = dosis * dias, "mL"
    elif tratamientos:
        resultado, unidad = abs(float(tratamientos.group(1)) - float(tratamientos.group(2))), "unidades"
    elif prototipo:
        piezas, precio = map(float, prototipo.groups())
        resultado, unidad = piezas * precio, "S/"
    else:
        raise ValueError("Caso de CIT sin datos reconocibles")
    resultado = formatear_numero(resultado)
    valores = [resultado, resultado + 1, resultado - 1, resultado + 2, resultado - 2]
    correcta = f"{resultado:g} {unidad}"
    return [f"{valor:g} {unidad}" for valor in valores], correcta

def opciones_fisica(problema):
    """Calcula cinco alternativas numéricas para cada situación física."""
    velocidad = re.search(r"bicicleta recorre (\d+) m en (\d+) s", problema)
    caida = re.search(r"herramienta desde (\d+) m", problema)
    fuerza = re.search(r"carrito de laboratorio de (\d+) kg.*?aceleración constante de (\d+) m/s²", problema)
    trabajo = re.search(r"fuerza horizontal de (\d+) N a lo largo de (\d+) m", problema)
    onda = re.search(r"frecuencia es (\d+) Hz y la longitud de onda (\d+) m", problema)
    if velocidad:
        distancia, tiempo = map(float, velocidad.groups())
        resultado, unidad = distancia / tiempo, "m/s"
    elif caida:
        altura = float(caida.group(1))
        resultado, unidad = math.sqrt(altura / 5), "s"
    elif fuerza:
        masa, aceleracion = map(float, fuerza.groups())
        resultado, unidad = masa * aceleracion, "N"
    elif trabajo:
        fuerza_aplicada, distancia = map(float, trabajo.groups())
        resultado, unidad = fuerza_aplicada * distancia, "J"
    elif onda:
        frecuencia, longitud = map(float, onda.groups())
        resultado, unidad = frecuencia * longitud, "m/s"
    else:
        raise ValueError("Enunciado de Física sin datos numéricos reconocibles")
    resultado = formatear_numero(resultado)
    valores = [resultado, resultado + 1, resultado - 1, resultado + 2, resultado - 2]
    correcta = f"{resultado:g} {unidad}"
    opciones = [f"{valor:g} {unidad}" for valor in valores]
    return opciones, correcta


def formatear_numero(numero):
    """Evita ceros decimales innecesarios en los resultados mostrados."""
    return int(numero) if float(numero).is_integer() else round(numero, 2)

def opciones_matematica(problema, indice):
    """Calcula respuestas directas para los problemas numéricos generados."""
    compra_descuento = re.search(
        r"compra (\d+) cajas con (\d+) unidades cada una a S/ (\d+) por unidad y recibe un descuento del (\d+) %",
        problema,
    )
    area_zona = re.search(
        r"terreno rectangular de (\d+) m de largo y (\d+) m de ancho se divide en (\d+) zonas iguales",
        problema,
    )
    ecuacion = re.search(
        r"(\d+) veces un número, aumentado en (\d+), es igual a (\d+)",
        problema,
    )
    proporcion_inversa = re.search(
        r"(\d+) personas terminan un proyecto en (\d+) horas.*?trabajan (\d+) personas",
        problema,
    )
    reparto = re.search(
        r"cantidad total de (\d+) respuestas se reparte en la razón (\d+):(\d+):1",
        problema,
    )
    suma = re.search(r"Calcula:?(?:\s+)(-?\d+) \+ \((-?\d+)\)", problema)
    resta = re.search(r"Calcula: \((-?\d+)\) - \((-?\d+)\)", problema)
    multiplicacion = re.search(r"Calcula: \((-?\d+)\) × (\d+)", problema)
    temperatura = re.search(r"temperatura de (-?\d+) °C (aumenta|baja) (\d+)", problema)
    ascensor = re.search(r"piso (-?\d+) y (sube|baja) (\d+) pisos", problema)
    if compra_descuento:
        cajas, unidades, precio, descuento = map(int, compra_descuento.groups())
        resultado = cajas * unidades * precio * (100 - descuento) / 100
    elif area_zona:
        largo, ancho, zonas = map(int, area_zona.groups())
        resultado = largo * ancho / zonas
    elif ecuacion:
        coeficiente, termino, total = map(int, ecuacion.groups())
        resultado = (total - termino) / coeficiente
    elif proporcion_inversa:
        personas, horas, nuevas_personas = map(int, proporcion_inversa.groups())
        resultado = personas * horas / nuevas_personas
    elif reparto:
        total, parte_a, parte_b = map(int, reparto.groups())
        resultado = total * parte_a / (parte_a + parte_b + 1)
    elif suma:
        resultado = int(suma.group(1)) + int(suma.group(2))
    elif resta:
        resultado = int(resta.group(1)) - int(resta.group(2))
    elif multiplicacion:
        resultado = int(multiplicacion.group(1)) * int(multiplicacion.group(2))
    elif temperatura:
        valor, operacion, cambio = temperatura.groups()
        resultado = int(valor) + (int(cambio) if operacion == "aumenta" else -int(cambio))
    elif ascensor:
        valor, operacion, cambio = ascensor.groups()
        resultado = int(valor) + (int(cambio) if operacion == "sube" else -int(cambio))
    else:
        compra = re.search(r"vende (\d+) productos a S/ (\d+)", problema)
        rectangulo = re.search(r"mide (\d+) cm por (\d+) cm", problema)
        tanque = re.search(r"tiene (\d+) L y se usan (\d+) L", problema)
        ahorro = re.search(r"ahorra S/ (\d+) durante (\d+) meses", problema)
        if compra:
            resultado = int(compra.group(1)) * int(compra.group(2))
        elif rectangulo:
            largo, ancho = map(int, rectangulo.groups())
            resultado = 2 * (largo + ancho)
        elif tanque:
            resultado = int(tanque.group(1)) - int(tanque.group(2))
        elif ahorro:
            resultado = int(ahorro.group(1)) * int(ahorro.group(2))
        else:
            resultado = indice + indice + 3
    if isinstance(resultado, float) and resultado.is_integer():
        resultado = int(resultado)
    correcta = f"Resultado: {resultado}"
    return [correcta, f"Resultado: {resultado + 1}", f"Resultado: {resultado - 1}"], correcta

def adaptar_ejercicios_subtema(ejercicios, area, subtema):
    """Añade una consigna de práctica coherente con el subtema."""
    practica = obtener_perfil_subtema(area, subtema)[4]
    return [
        f"{ejercicio} Tema aplicado: {subtema}. {practica}"
        for ejercicio in ejercicios
    ]


def construir_teoria_subtema(subtema, area, etapa):
    """Construye una explicación tipo libro con concepto, partes, método y ejemplo."""
    introduccion, partes, estructura, ejemplo, _ = obtener_perfil_subtema(area, subtema)
    return (
        f"**¿Qué es este subtema?**\n{introduccion}\n\n"
        f"**¿De qué trata?**\nEste subtema desarrolla el concepto de {subtema} y explica cómo se reconoce y se aplica en situaciones concretas de {area}.\n\n"
        f"**Partes principales**\nEste contenido se compone de {partes}. Comprender cada elemento permite explicar el subtema y aplicarlo correctamente.\n\n"
        f"**¿Cómo se resuelve o se estudia?**\n{estructura}.\n\n"
        f"**Ejemplo explicado**\n{ejemplo}"
    )


def obtener_imagen_subtema(area, subtema):
    """Construye una imagen estable usando el área y subtema exactos."""
    consulta = quote(f"{area} {subtema}", safe="")
    identificador = int(md5(f"{area}:{subtema}".encode("utf-8")).hexdigest()[:8], 16)
    return f"https://loremflickr.com/1200/700/{consulta}?lock={identificador}"


def mostrar_subtema(subtema, area, etapa, clave):
    """Presenta conocimiento, ejercicios y examen final para un subtema."""
    guia = GUIAS_AREA[area]
    teoria_subtema = construir_teoria_subtema(subtema, area, etapa)
    st.markdown(f"##### Subtema: {subtema}")
    clase, video, solucionario, banco_preguntas, evaluacion = st.tabs(
        ["📘 Clase", "▶️ Video", "🧩 Solucionario", "📚 Banco de preguntas", "📝 Examen"]
    )
    with clase:
        st.markdown("**Teoría detallada**")
        st.image(
            obtener_imagen_subtema(area, subtema),
            caption=f"Ejemplo visual de {subtema}",
            use_container_width=True,
        )
        st.markdown("**Definición**")
        st.write(teoria_subtema)
        st.markdown("**Ruta guiada del contenido**")
        st.write(f"➊ Reconoce el concepto **{subtema}** y sus palabras clave.")
        st.write("➋ Separa los datos, elementos o evidencias que aparecen en el caso.")
        st.write("➌ Relaciona esos elementos con la definición y sus propiedades.")
        st.write("➍ Formula una conclusión y verifica que sea coherente con el concepto.")
        st.markdown("**Elementos y aplicación**")
        st.write(
            f"En {etapa}, **{subtema}** aparece cuando es necesario {EJEMPLOS_AREA[area]}. "
            "Sus elementos se interpretan según la definición y sus propiedades "
            "permiten comprobar la conclusión. "
            f"En {area}, esto se concreta en {guia['teoria'].lower()}"
        )
        if subtema == "Números enteros":
            st.markdown("**Principios y fórmulas**")
            st.latex(r"\mathbb{Z} = \{..., -2, -1, 0, 1, 2, ...\}")
            st.latex(r"a - b = a + (-b) \qquad |a| = \text{distancia de } a \text{ al cero}")
            st.write("Ejemplo 1: (-7) + 3 = -(7 - 3) = -4, porque los signos son distintos y 7 es mayor que 3.")
            st.write("Ejemplo 2: (-4) - (-9) = (-4) + 9 = 5, porque restar -9 equivale a sumar 9.")
            st.write("Ejemplo 3: (-6) × (-2) = 12, porque dos signos negativos producen un resultado positivo.")
        st.markdown("**Idea clave**")
        st.write(
            f"La idea central de **{subtema}** se conserva aunque cambien los datos, "
            "los materiales o el contexto del problema."
        )
        st.markdown("**Ejercicios de aplicación**")
        if subtema == "Números enteros":
            st.write("1. Ordena de menor a mayor: -8, 3, 0, -2, 6.")
            st.write("2. Calcula: (-12) + 7, 9 - (-4) y (-5) - 8.")
            st.write("3. Resuelve: una temperatura de -3 °C aumenta 11 °C. ¿Cuál es la temperatura final?")
            st.write("4. Calcula y justifica el signo: (-6) × 4, (-24) ÷ (-3) y 18 ÷ (-6).")
        else:
            for indice, ejercicio in enumerate(guia["ejercicios"], start=1):
                st.write(f"{indice}. {ejercicio} Tema: **{subtema}**.")
        st.markdown("**Ejemplo práctico**")
        st.write(
            f"Ejemplo resuelto: en {etapa}, {guia['ejemplo']} "
            f"El caso se relaciona con **{subtema}** porque usa sus elementos y propiedades."
        )
        st.markdown("**Caso resuelto**")
        st.write(
            f"Caso: una situación de {area} contiene información sobre **{subtema}**. "
            "Identificación: se localizan los datos relevantes. Aplicación: se "
            "relacionan con la definición. Resultado: se expresa una conclusión "
            "completa y se comprueba con las propiedades del concepto."
        )
    with video:
        st.markdown(f"**Video conceptual: {subtema}**")
        st.info(
            f"Recurso visual sugerido para {etapa}: representa **{subtema}** con "
            f"imágenes, palabras clave y un caso de {area}."
        )
        st.progress(0, text="Video pendiente de incorporar")
    with solucionario:
        st.markdown("**Solucionario de la práctica**")
        if subtema == "Números enteros":
            st.write("1. Orden: -8 < -2 < 0 < 3 < 6.")
            st.write("2. Operaciones: (-12) + 7 = -5; 9 - (-4) = 13; (-5) - 8 = -13.")
            st.write("3. Temperatura: -3 °C + 11 °C = 8 °C.")
            st.write("4. Productos y cocientes: (-6) × 4 = -24; (-24) ÷ (-3) = 8; 18 ÷ (-6) = -3.")
        else:
            for indice, solucion in enumerate(guia["soluciones"], start=1):
                st.write(f"{indice}. {solucion}")
    with banco_preguntas:
        st.markdown(f"**100 preguntas de práctica sobre {subtema}**")
        st.write("Estas preguntas se generan exclusivamente para el subtema seleccionado.")
        banco = generar_banco_ejercicios(area, subtema, etapa, st.session_state.nivel)
        for indice, ejercicio in enumerate(banco, start=1):
            st.write(f"{indice}. {ejercicio}")
    with evaluacion:
        st.markdown(f"**Examen final del subtema: {subtema}**")
        st.write("Selecciona una opción en cada problema. Cada examen combina problemas diferentes del subtema.")
        clave_examen = f"examen_{clave}"
        if st.button("Generar otro examen", key=f"nuevo_{clave}"):
            st.session_state.pop(clave_examen, None)
            st.session_state[f"{clave_examen}_version"] = (
                st.session_state.get(f"{clave_examen}_version", 0) + 1
            )
            st.rerun()
        banco_examen = obtener_banco_examen(
            clave, area, subtema, etapa, st.session_state.nivel
        )
        niveles = [("🟢 Básico", banco_examen[:15]), ("🟡 Intermedio", banco_examen[15:35]), ("🔴 Avanzado", banco_examen[35:])]
        with st.form(key=f"formulario_examen_{clave}"):
            respuestas_banco = []
            correctas_banco = []
            numero_problema = 1
            for nombre_nivel, problemas in niveles:
                st.markdown(f"**{nombre_nivel}**")
                for problema, opciones, correcta in problemas:
                    respuestas_banco.append(
                        st.radio(
                            f"Problema {numero_problema}: {problema}",
                            opciones,
                            index=None,
                            key=f"banco_examen_{numero_problema}_{clave}_{obtener_version_examen(clave)}",
                        )
                    )
                    correctas_banco.append(correcta)
                    numero_problema += 1
            enviar_examen = st.form_submit_button("Calificar las 50 preguntas")
        if enviar_examen:
            puntaje = 0
            for respuesta, correcta in zip(respuestas_banco, correctas_banco):
                if respuesta == correcta:
                    puntaje += 1
                    st.session_state.racha_respuestas += 1
                else:
                    st.session_state.racha_respuestas = 0

            promedio_aprobatorio = len(respuestas_banco) / 2
            aprobado = puntaje >= promedio_aprobatorio
            puntos_tema = 50
            if aprobado and clave not in st.session_state.temas_completados:
                st.session_state.puntos += puntos_tema
                st.session_state.temas_completados.add(clave)
                st.session_state.racha_temas += 1
                mensaje_puntos = (
                    f" Ganaste +{puntos_tema} puntos por terminar el tema. "
                    f"Racha de temas: {st.session_state.racha_temas}."
                )
            elif not aprobado:
                mensaje_puntos = (
                    f" No ganaste puntos: necesitas al menos {promedio_aprobatorio:g} "
                    "respuestas correctas para aprobar el tema."
                )
            else:
                mensaje_puntos = " Este tema ya había otorgado sus puntos."
            st.success(
                f"Resultado del examen: {puntaje} de {len(respuestas_banco)} respuestas correctas."
                f" Racha de respuestas: {st.session_state.racha_respuestas}.{mensaje_puntos}"
            )

def completar_subtemas_por_tema():
    """Garantiza diez subtemas relacionados con cada tema y cada área."""
    for datos_nivel in CONTENIDOS.values():
        for contenidos_etapa in datos_nivel.values():
            for area, temas in contenidos_etapa.items():
                complementarios = SUBTEMAS_COMPLEMENTARIOS[area]
                for tema, subtemas in temas.items():
                    indice = 0
                    while len(subtemas) < 10:
                        complemento = complementarios[indice % len(complementarios)]
                        subtema = f"{complemento} de {tema}"
                        if subtema not in subtemas:
                            subtemas.append(subtema)
                        indice += 1

def obtener_progreso_nivel(nivel):
    """Calcula el avance del nivel según los subtemas aprobados."""
    total_subtemas = sum(
        len(subtemas)
        for contenidos_etapa in CONTENIDOS[nivel].values()
        for subtemas_por_tema in contenidos_etapa.values()
        for subtemas in subtemas_por_tema.values()
    )
    temas_aprobados = sum(
        clave.startswith(f"{nivel}_") for clave in st.session_state.temas_completados
    )
    porcentaje = (temas_aprobados / total_subtemas * 100) if total_subtemas else 0
    return temas_aprobados, total_subtemas, porcentaje
